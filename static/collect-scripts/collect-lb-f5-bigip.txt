# Keystone collect list - F5 BIG-IP (TMOS 12.1 - 17.x)
# Log in with a read-only account (the Auditor or Guest role is enough), run the commands from the advanced shell (bash) or from tmsh, save the whole terminal buffer and upload it to the Load Balancer Analyzer.
# Every command below only reads. 'tmsh list' prints only the values that differ from the default; the all-properties lines add the defaults the analyzer needs to judge inherited SSL options.

tmsh show sys version
tmsh show sys hardware

# Whole configuration: LTM, net, sys, auth and cm objects (this one command replaces bigip.conf + bigip_base.conf).
tmsh list

# Effective values including defaults for the objects the findings depend on.
tmsh list ltm profile client-ssl all-properties
tmsh list ltm profile server-ssl all-properties
tmsh list ltm virtual all-properties
tmsh list sys file ssl-cert
tmsh list net self-allow

# Live state.
tmsh show cm sync-status
tmsh show cm failover-status
tmsh show ltm virtual
tmsh show ltm pool
tmsh show ltm pool members
