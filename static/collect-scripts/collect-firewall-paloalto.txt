# Keystone collect list - Palo Alto Networks PAN-OS
# Run from the CLI (operational mode). The set format is the easiest to analyse; an XML export of the running configuration also works.
set cli pager off
set cli config-output-format set
configure
show
exit
show system info
show high-availability state
show routing route
show vpn ike-sa
show vpn ipsec-sa

# Neighbors and ARP, for the Topology Builder (not used by the Firewall Analyzer). LLDP must be enabled on the interfaces (Network > LLDP). If your PAN-OS version does not accept the first command, read the neighbors in the web interface under Network > LLDP > Peers.
show lldp neighbors all
show arp all
